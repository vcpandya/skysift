\documentclass{aastex7}
% The frame of a Research Note of the AAS (RNAAS), as SkySift fills it. The note itself starts at the begin-document
% line below this frame and is written by skysift.pack.render_note from the candidate's dossier.
%
% RNAAS rules (AAS research note preparation guidelines, read 21 October 2026; check them again before you submit):
% at most 1,500 words including the title, headings, figure caption and references, of which up to 150 are the
% required abstract; one figure or one table, not both. The note is reviewed by the editor, not refereed, and gets a
% DOI when it is published. The AAS prefers AASTeX v7 (aastex7.cls), which has no separate RNAAS class option: the
% journal is chosen at submission. UAT keywords can also be chosen in the submission form.
%
% Compile with pdflatex and aastex7.cls (on Overleaf as the AAS template, or from the AAS journals site). The figure
% is figure1.pdf when the pack was built with matplotlib; otherwise convert figure1.svg to PDF first
% (Inkscape: inkscape figure1.svg --export-type=pdf).
\usepackage{graphicx}
% AASTeX 7.0.1 prints "Draft version <today>" (its default \journalinfo) and a "Typeset using" line above the
% title. A Research Note is published from the author's own PDF, so the first is blanked with the class's own
% command and the second is patched out of the default style's header; a class without that macro leaves the note
% as it is, and the log says which (a SKYSIFT line).
\journalinfo{}
\usepackage{etoolbox}
\makeatletter
\ifcsname frontmatter@title@above\endcsname
\patchcmd{\frontmatter@title@above}{{\footnotesize Typeset using \LaTeX\ {\bf default} style in AASTeX7.0.1}}{}{\typeout{SKYSIFT: typeset line removed}}{\typeout{SKYSIFT: typeset line not patched}}
\fi
\makeatother
